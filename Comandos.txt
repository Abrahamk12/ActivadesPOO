private void btnMostrarDialogo_Click(object sender, EventArgs e)
{
    DialogResult resultado = MessageBox.Show("¿Estás seguro de que quieres realizar esta acción?", "Confirmación", MessageBoxButtons.YesNoCancel, MessageBoxIcon.Question);

    if (resultado == DialogResult.Yes)
    {
        MessageBox.Show("Acción realizada.", "Resultado", MessageBoxButtons.OK, MessageBoxIcon.Information);
    }
    else if (resultado == DialogResult.No)
    {
        MessageBox.Show("Acción cancelada.", "Resultado", MessageBoxButtons.OK, MessageBoxIcon.Warning);
    }
    else
    {
        MessageBox.Show("Acción abortada.", "Resultado", MessageBoxButtons.OK, MessageBoxIcon.Exclamation);
    }
}
